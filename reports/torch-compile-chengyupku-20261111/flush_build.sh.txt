#!/bin/bash
set -eu
cd ${FLUSH_WORK}
source "${BASELINE_ACTIVATE}"
FLUSH_CANN92=${FLUSH_CANN92_ROOT}
FLUSH_COMPILER_BIN=$FLUSH_CANN92/tools/bisheng_compiler/bin
FLUSH_ASC=$FLUSH_CANN92/aarch64-linux/asc
FLUSH_ACL91=${BASELINE_CANN91_ROOT}/aarch64-linux/include
FLUSH_TILELANG=$PWD/tilelang-0a2fb6884004b73362f584e10c35e09d9bed3a6e
mkdir build01
set -x
ASCEND_HOME_PATH="$FLUSH_CANN92" "$FLUSH_COMPILER_BIN/bisheng" -O2 -fPIC -std=c++20 -mllvm -cce-aicore-dcpreload-args=false --npu-arch=dav-3510 --cce-aicore-only -I "$FLUSH_TILELANG/src" -I "$FLUSH_ASC/include" -I "$FLUSH_ASC" -I "$FLUSH_ACL91" device_kernel.asc -o build01/flush.rel.o
"$FLUSH_COMPILER_BIN/ld.lld" -m aicorelinux -Ttext 0 --no-mmap-output-file build01/flush.rel.o -o build01/flush.aibin
"$CXX" -O2 -std=c++17 -shared -fPIC loader.cpp -I "$FLUSH_ACL91" -L ${BASELINE_CANN91_ROOT}/aarch64-linux/lib64 -lascendcl -o build01/flush.so
sha256sum build01/flush.aibin build01/flush.so "$FLUSH_COMPILER_BIN/bisheng" "$FLUSH_COMPILER_BIN/ld.lld"
